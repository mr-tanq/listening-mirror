}); self.addEventListener("fetch", (event) => { const req = event.request; const url = new URL(req.url); if (url.origin !== self.location.origin) return; if (req.mode === "navigate") { event.respondWith( fetch(req).catch(() => caches.match("./")) ); return; } event.respondWith( caches.match(req).then((cached) => cached || fetch(req).then((res) => { const copy = res.clone(); caches.open(CACHE).then((c) => c.put(req, copy)).catch(()=>{}); return res; })) ); });
